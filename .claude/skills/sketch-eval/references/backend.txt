title "How our backend looks"
app "Mobile app"
gateway "API gateway"
users "Users service"
orders "Orders service"
users_db "Users DB (Postgres)"
orders_db "Orders DB (Postgres)"
redis "Redis cache"
kafka "Kafka topic"
worker "Worker"
sendgrid "SendGrid"
app -> gateway "requests"
gateway -> users "forwards"
gateway -> orders "forwards"
users -> users_db "own db"
users -> redis "cache profiles"
orders -> orders_db "own db"
orders -> kafka "order placed"
kafka -> worker "reads"
worker -> sendgrid "sends emails"
